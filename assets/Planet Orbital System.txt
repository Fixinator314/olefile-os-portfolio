/* =================================
   ORBIT CLICK (FIXED)
================================= */

const planets = document.querySelectorAll(".planet");
const panels = document.querySelectorAll(".orbit-panel");

planets.forEach((planet) => {
  planet.addEventListener("click", () => {
    const targetId = planet.dataset.panel;
    const targetPanel = document.getElementById(targetId);

    gsap.to(planets, {
      scale: 1,
      duration: 0.3,
    });

    gsap.to(planet, {
      scale: 1.2,
      duration: 0.3,
    });

    if (!targetPanel) {
      console.error(`Panel with id "${targetId}" not found!`);
      return;
    }

    // Remove active from all planets first
    planets.forEach((p) => p.classList.remove("active-planet"));
    planet.classList.add("active-planet");

    // Remove active from all panels
    panels.forEach((panel) => {
      panel.classList.remove("active-panel");
    });

    // Force a reflow before adding the new active class
    // This ensures the transition restarts properly
    void targetPanel.offsetWidth;

    // Add active to target panel
    targetPanel.classList.add("active-panel");
  });
});
